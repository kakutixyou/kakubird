-- ============================================================
-- infra/migrations/001_initial_schema.sql
-- 初期スキーマ。supabase db push で適用する。
-- ============================================================

-- -----------------------------------------------------------
-- organizations テーブル（テナント）
-- -----------------------------------------------------------
CREATE TABLE organizations (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name          TEXT NOT NULL,
  slug          TEXT NOT NULL UNIQUE,
  is_pro_plan   BOOLEAN NOT NULL DEFAULT false,
  plan_expires_at TIMESTAMPTZ,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- -----------------------------------------------------------
-- departments テーブル（部署）
-- -----------------------------------------------------------
CREATE TABLE departments (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  org_id     UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
  name       TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- -----------------------------------------------------------
-- users テーブル（auth.users と連携）
-- Supabase Auth の auth.users.id と同一の UUID を使う。
-- -----------------------------------------------------------
CREATE TABLE users (
  id              UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  org_id          UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
  department_id   UUID REFERENCES departments(id) ON DELETE SET NULL,
  email           TEXT NOT NULL,
  display_name    TEXT NOT NULL DEFAULT '',
  role            TEXT NOT NULL DEFAULT 'member'
                  CHECK (role IN ('owner','admin','manager','member','viewer')),
  avatar_url      TEXT,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- -----------------------------------------------------------
-- audit_logs テーブル（簡略版：主要操作のみ記録）
-- -----------------------------------------------------------
CREATE TABLE audit_logs (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  org_id     UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
  user_id    UUID REFERENCES users(id) ON DELETE SET NULL,
  action     TEXT NOT NULL,   -- 例: "users.invite", "users.remove"
  target_id  TEXT,            -- 操作対象の ID
  meta       JSONB,           -- 追加情報（任意）
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- -----------------------------------------------------------
-- RLS 有効化（Row Level Security）
-- 各テナントは自分の org_id のデータしか見えない。
-- -----------------------------------------------------------
ALTER TABLE organizations ENABLE ROW LEVEL SECURITY;
ALTER TABLE departments   ENABLE ROW LEVEL SECURITY;
ALTER TABLE users         ENABLE ROW LEVEL SECURITY;
ALTER TABLE audit_logs    ENABLE ROW LEVEL SECURITY;

-- organizations: 自分の org のみ読み取り可
CREATE POLICY "org_select" ON organizations
  FOR SELECT USING (
    id IN (SELECT org_id FROM users WHERE id = auth.uid())
  );

-- departments: 同一 org のみ
CREATE POLICY "dept_select" ON departments
  FOR SELECT USING (
    org_id IN (SELECT org_id FROM users WHERE id = auth.uid())
  );

CREATE POLICY "dept_insert" ON departments
  FOR INSERT WITH CHECK (
    org_id IN (SELECT org_id FROM users WHERE id = auth.uid())
  );

CREATE POLICY "dept_update" ON departments
  FOR UPDATE USING (
    org_id IN (SELECT org_id FROM users WHERE id = auth.uid())
  );

-- users: 同一 org のみ
CREATE POLICY "users_select" ON users
  FOR SELECT USING (
    org_id IN (SELECT org_id FROM users WHERE id = auth.uid())
  );

CREATE POLICY "users_update" ON users
  FOR UPDATE USING (
    org_id IN (SELECT org_id FROM users WHERE id = auth.uid())
  );

CREATE POLICY "users_delete" ON users
  FOR DELETE USING (
    org_id IN (SELECT org_id FROM users WHERE id = auth.uid())
  );

-- audit_logs: 同一 org のみ読み取り
CREATE POLICY "audit_select" ON audit_logs
  FOR SELECT USING (
    org_id IN (SELECT org_id FROM users WHERE id = auth.uid())
  );

-- -----------------------------------------------------------
-- インデックス（パフォーマンス最適化）
-- -----------------------------------------------------------
CREATE INDEX idx_users_org_id        ON users(org_id);
CREATE INDEX idx_departments_org_id  ON departments(org_id);
CREATE INDEX idx_audit_logs_org_id   ON audit_logs(org_id);
CREATE INDEX idx_audit_logs_user_id  ON audit_logs(user_id);
